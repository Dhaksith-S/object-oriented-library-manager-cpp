cmake_minimum_required(VERSION 3.16)
project(ObjectOrientedLibraryManager VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(library_core src/Book.cpp src/Member.cpp src/Loan.cpp src/Date.cpp
    src/LibraryManager.cpp src/StorageManager.cpp)
target_include_directories(library_core PUBLIC include)
if(MSVC)
    target_compile_options(library_core PUBLIC /W4 /permissive-)
else()
    target_compile_options(library_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(library_manager src/main.cpp)
target_link_libraries(library_manager PRIVATE library_core)
include(CTest)
if(BUILD_TESTING)
    foreach(suite models library storage)
        add_executable(test_${suite} tests/test_${suite}.cpp)
        target_include_directories(test_${suite} PRIVATE tests)
        target_link_libraries(test_${suite} PRIVATE library_core)
        add_test(NAME ${suite} COMMAND test_${suite} ${CMAKE_CURRENT_SOURCE_DIR}/data/sample_library.csv)
    endforeach()
    find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py
        $<TARGET_FILE:library_manager> ${CMAKE_CURRENT_SOURCE_DIR}/data/sample_library.csv)
endif()
